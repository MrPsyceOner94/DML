name: Run Matchup Extractor

on:
  workflow_dispatch:
    inputs:
      url:
        description: 'Matchup URL to fetch'
        required: true
        default: 'https://fantasy.nrl.com/draft/league/60018/matchup/68/1392/24'
      use_team_creds:
        description: 'Use stored team credentials (true/false)'
        required: false
        default: 'false'
      league:
        description: 'League id (when using team creds)'
        required: false
        default: ''
      team:
        description: 'Team source id (when using team creds)'
        required: false
        default: ''

jobs:
  extract:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: '3.10'

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

      - name: Install Playwright browsers
        run: |
          python -m playwright install --with-deps

      - name: Run extractor
        env:
          DATABASE_URL: ${{ secrets.DATABASE_URL }}
          CRED_ENCRYPTION_KEY: ${{ secrets.CRED_ENCRYPTION_KEY }}
          ADMIN_USERNAME: ${{ secrets.ADMIN_USERNAME }}
          ADMIN_PASSWORD: ${{ secrets.ADMIN_PASSWORD }}
          NRL_LOGIN_URL: ${{ secrets.NRL_LOGIN_URL }}
        run: |
          mkdir -p output
          # build command depending on inputs
          if [ "${{ github.event.inputs.use_team_creds }}" = "true" ]; then
            python fetch_matchup_playwright.py "${{ github.event.inputs.url }}" --outdir ./output --use-team-creds --league "${{ github.event.inputs.league }}" --team "${{ github.event.inputs.team }}"
          else
            python fetch_matchup_playwright.py "${{ github.event.inputs.url }}" --outdir ./output
          fi

      - name: Upload artifact
        uses: actions/upload-artifact@v4
        with:
          name: matchup-output
          path: output/**
